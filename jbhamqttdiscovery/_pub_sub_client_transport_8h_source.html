93 bool connect(const char* clientId = nullptr) override {
94 if (!clientId || !*clientId) {

96 return false;
97 }
98 if (clientId != clientIdStr.c_str()) {
99 clientIdStr = clientId;
100 }
101 takeReconnectRequest(); // this is the attempt a pending request asked for
102
103 const char* user = userStr.empty() ? nullptr : userStr.c_str();
104 const char* pass = passStr.empty() ? nullptr : passStr.c_str();
105 const char* willTopic = willTopicStr.empty() ? nullptr : willTopicStr.c_str();
106 const char* willMessage = willTopic ? willPayloadStr.c_str() : nullptr;
107
108 if (log) log->info("PubSub connecting as %s", clientId);
109 bool ok = client.connect(clientId, user, pass, willTopic, willQos, willRetain, willMessage);
110 if (!ok) {
111 if (log) log->error("PubSub connect FAILED, state=%d", client.state());

113 return false;
114 }
115
116 // Report the connection now, so the next tick() doesn't have to detect it
117 reconnect.reset(millis());
118 wasConnected = true;

135 bool publish(const char* topic,
136 const uint8_t* payload,
137 size_t len,
138 bool retained,
139 uint8_t /*qos*/) override {
140 if (payload == nullptr) {
141 len = 0;
142 }
143
144 size_t needed = PACKET_OVERHEAD + strlen(topic) + len;
145 if (needed > client.getBufferSize()) {
146 if (log) log->info("PubSub buffer too small (%u < %u), growing it", (unsigned)client.getBufferSize(), (unsigned)needed);
147 if (!ensureBufferSize(needed)) {
148 if (log) log->error("PubSub publish FAILED topic=%s, could not grow buffer to %u bytes", topic, (unsigned)needed);
149 return false;
150 }
151 }
152
153 if (log) log->debug("PubSub publish topic=%s len=%u retained=%d", topic,
154 (unsigned)len, retained ? 1 : 0);
155
156 bool ok = client.publish(topic,
157 payload,
158 static_cast<unsigned int>(len),
159 retained);
160
161 if (!ok) {
162 if (log) log->error("PubSub publish FAILED topic=%s", topic);
163 } else {
164 if (log) log->debug("PubSub publish OK topic=%s", topic);
165 }
166 return ok;

230 void tick() override {
231 bool nowConnected = client.connected();
232 uint32_t now = millis();
233
234 if (nowConnected && !wasConnected) {
235 reconnect.reset(now);
236 if (cb) {
237 cb(ctx);
238 }
239 } else if (!nowConnected && wasConnected) {
240 if (log) log->warning("PubSub connection lost, state=%d", client.state());
241 reconnect.reset(now);
242 }
243 wasConnected = nowConnected;
244
245 if (takeReconnectRequest() && !nowConnected) {
246 reconnect.reset(now);
247 }
248
249 if (!nowConnected && !clientIdStr.empty() && reconnect.due(now)) {
250 if (log) log->info("PubSub reconnecting");
251 connect(clientIdStr.c_str());
252 }
253 }
