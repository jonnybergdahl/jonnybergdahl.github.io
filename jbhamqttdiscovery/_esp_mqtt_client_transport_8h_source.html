224 void tick() override {
225 if (callLoop) {
226 client.loop();
227 }
228 if (!started) {
229 return;
230 }
231 uint32_t now = millis();
232
233 if (disconnectEvent) {
234 disconnectEvent = false;
235 if (droppedSession) {
236 if (log) log->warning("espMqtt connection lost, reason %u", (unsigned)lastReason);
237 reconnect.reset(now); // try again right away
238 } else if (!attemptFailed) {
239 if (log) log->error("espMqtt connect FAILED, reason %u", (unsigned)lastReason);
240 attemptFailed = true;
241 reconnect.failed(now); // wait before the next attempt
242 }
243 }
244
245 if (takeReconnectRequest() && !isConnected) {
246 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(STALE_ATTEMPT_MS)) {
247 if (log) log->info("espMqtt reconnect requested, dropping the current attempt");
248 connecting = false;
249 attemptFailed = true;
250 client.disconnect(true);
251 }
252 reconnect.reset(now);
253 }
254
255 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(CONNECT_TIMEOUT_MS)) {
256 // Safety net: espMqttClient normally reports a failed attempt itself
257 if (log) log->error("espMqtt connect timed out, resetting the client");
258 connecting = false;
259 client.disconnect(true);
260 if (!attemptFailed) {
261 attemptFailed = true;
262 reconnect.failed(now);
263 }
264 }
265
266 // Only start a new attempt once the client is fully disconnected
267 if (!isConnected && !connecting && client.disconnected() && reconnect.due(now)) {
268 if (log) log->info("espMqtt reconnecting");
269 startAttempt();
270 }
271 }
272
273private:
274 void startAttempt() {
275 if (log) log->info("espMqtt connecting");
276 attemptStartedAt = millis();
277 attemptFailed = false;
278 // Set before connect(): onConnect may fire in another task before connect() returns
279 connecting = true;
280 if (!client.connect()) {
281 connecting = false;
282 // Not in a state to connect yet (e.g. still disconnecting); tick() retries later
283 attemptFailed = true;
284 reconnect.failed(attemptStartedAt);
285 }
286 }
