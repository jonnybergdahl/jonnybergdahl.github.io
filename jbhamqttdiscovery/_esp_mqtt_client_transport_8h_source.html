137 bool publish(const char* topic, const uint8_t* payload, size_t len, bool retained, uint8_t qos) override {
138 if (!isConnected) {
139 // espMqttClient would queue it, but HaDiscovery stores and re-sends on connect itself

144 uint16_t pid = client.publish(topic, qos, retained, payload ? payload : &empty, payload ? len : 0);
145 if (pid == 0) {
146#if defined(ESP8266) && defined(EMC_MIN_FREE_MEMORY)

150#else

152 if (log) log->error("espMqtt publish FAILED topic=%s len=%u", topic, (unsigned)len);
153#endif
154 return false;
155 }
156 if (log) log->debug("espMqtt publish OK topic=%s len=%u pid=%u", topic, (unsigned)len, (unsigned)pid);

182 void setOnMessage(MessageCallback cb_, void* ctx_) override {
183 msgCb = cb_;
184 msgCtx = ctx_;
185 client.onMessage([this](const espMqttClientTypes::MessageProperties& /*properties*/, const char* topic,
186 const uint8_t* payload, size_t len, size_t index, size_t total) {
187 if (index == 0 && len == total) {
188 if (msgCb) {
189 msgCb(msgCtx, topic, payload, len);
190 }
191 return;
192 }
193 switch (assembler.add(topic, payload, len, index, total)) {
194 case MessageAssembler::Result::Complete:
195 if (msgCb) {
196 msgCb(msgCtx, assembler.topic().c_str(),
197 reinterpret_cast<const uint8_t*>(assembler.payload().data()), assembler.payload().size());
198 }
199 assembler.release();
200 break;
201 case MessageAssembler::Result::TooLarge:
202 if (log) log->warning("espMqtt message on %s is %u bytes, more than %u, ignored",
203 topic, (unsigned)total, (unsigned)assembler.maxSize());
204 break;
205 case MessageAssembler::Result::Broken:
206 if (log) log->warning("espMqtt message on %s arrived out of order, ignored", topic);
207 break;
208 default:
209 break;
210 }
211 });

265 void tick() override {
266 if (callLoop) {
267 client.loop();
268 }
269 if (!started) {
270 return;
271 }
272 uint32_t now = millis();
273
274 if (disconnectEvent) {
275 disconnectEvent = false;
276 if (droppedSession) {
277 if (log) log->warning("espMqtt connection lost, reason %u", (unsigned)lastReason);
278 reconnect.reset(now); // try again right away
279 } else if (!attemptFailed) {
280 if (log) log->error("espMqtt connect FAILED, reason %u", (unsigned)lastReason);
281 attemptFailed = true;
282 reconnect.failed(now); // wait before the next attempt
283 }
284 }
285
286 if (takeReconnectRequest() && !isConnected) {
287 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(STALE_ATTEMPT_MS)) {
288 if (log) log->info("espMqtt reconnect requested, dropping the current attempt");
289 connecting = false;
290 attemptFailed = true;
291 client.disconnect(true);
292 }
293 reconnect.reset(now);
294 }
295
296 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(CONNECT_TIMEOUT_MS)) {
297 // Safety net: espMqttClient normally reports a failed attempt itself
298 if (log) log->error("espMqtt connect timed out, resetting the client");
299 connecting = false;
300 client.disconnect(true);
301 if (!attemptFailed) {
302 attemptFailed = true;
303 reconnect.failed(now);
304 }
305 }
306
307 // Only start a new attempt once the client is fully disconnected
308 if (!isConnected && !connecting && client.disconnected() && reconnect.due(now)) {
309 if (log) log->info("espMqtt reconnecting");
310 startAttempt();
311 }
312 }
313
314private:
315 void startAttempt() {
316 if (log) log->info("espMqtt connecting");
317 attemptStartedAt = millis();
318 attemptFailed = false;
319 // Set before connect(): onConnect may fire in another task before connect() returns
320 connecting = true;
321 if (!client.connect()) {
322 connecting = false;
323 // Not in a state to connect yet (e.g. still disconnecting); tick() retries later
324 attemptFailed = true;
325 reconnect.failed(attemptStartedAt);
326 }
327 }
