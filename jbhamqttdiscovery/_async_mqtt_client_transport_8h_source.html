117 void tick() override {
118 if (!started) {
119 return;
120 }
121 uint32_t now = millis();
122
123 if (disconnectEvent) {
124 disconnectEvent = false;
125 if (droppedSession) {
126 if (log) log->warning("Async connection lost");
127 reconnect.reset(now); // try again right away
128 } else if (!attemptFailed) {
129 if (log) log->error("Async connect FAILED");
130 attemptFailed = true;
131 reconnect.failed(now); // wait before the next attempt
132 }
133 }
134
135 if (takeReconnectRequest() && !isConnected) {
136 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(STALE_ATTEMPT_MS)) {
137 // An attempt started while the network was down may never call back,
138 // so drop it instead of waiting for the timeout. A fresh one is left alone.
139 if (log) log->info("Async reconnect requested, dropping the current attempt");
140 connecting = false;
141 attemptFailed = true; // a late onDisconnect for it isn't a new failure
142 client.disconnect(true);
143 }
144 reconnect.reset(now);
145 }
146
147 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(CONNECT_TIMEOUT_MS)) {
148 // No onConnect or onDisconnect came: reset the client so connect() works again
149 if (log) log->error("Async connect timed out, resetting the client");
150 connecting = false;
151 client.disconnect(true);
152 if (!attemptFailed) {
153 attemptFailed = true;
154 reconnect.failed(now);
155 }
156 }
157
158 if (!isConnected && !connecting && reconnect.due(now)) {
159 if (log) log->info("Async reconnecting");
160 startAttempt();
161 }
162 }

222 void setOnMessage(MessageCallback cb_, void* ctx_) override {
223 msgCb = cb_;

225
226 client.onMessage([this](char* topic, char* payload, AsyncMqttClientMessageProperties /*properties*/,
227 size_t len, size_t index, size_t total) {
228 const uint8_t* data = reinterpret_cast<const uint8_t*>(payload);
229 if (index == 0 && len == total) {
230 deliver(topic, data, len);
231 return;
232 }
233 switch (assembler.add(topic, data, len, index, total)) {
234 case MessageAssembler::Result::Complete:
235 deliver(assembler.topic().c_str(), reinterpret_cast<const uint8_t*>(assembler.payload().data()),
236 assembler.payload().size());
237 assembler.release();
238 break;
239 case MessageAssembler::Result::TooLarge:
240 if (log) log->warning("Async message on %s is %u bytes, more than %u, ignored",
241 topic, (unsigned)total, (unsigned)assembler.maxSize());
242 break;
243 case MessageAssembler::Result::Broken:
244 if (log) log->warning("Async message on %s arrived out of order, ignored", topic);
245 break;
246 default:
247 break;
248 }
249 });
