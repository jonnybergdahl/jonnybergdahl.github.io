116 void tick() override {
117 if (!started) {
118 return;
119 }
120 uint32_t now = millis();
121
122 if (disconnectEvent) {
123 disconnectEvent = false;
124 if (droppedSession) {
125 if (log) log->warning("Async connection lost");
126 reconnect.reset(now); // try again right away
127 } else if (!attemptFailed) {
128 if (log) log->error("Async connect FAILED");
129 attemptFailed = true;
130 reconnect.failed(now); // wait before the next attempt
131 }
132 }
133
134 if (takeReconnectRequest() && !isConnected) {
135 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(STALE_ATTEMPT_MS)) {
136 // An attempt started while the network was down may never call back,
137 // so drop it instead of waiting for the timeout. A fresh one is left alone.
138 if (log) log->info("Async reconnect requested, dropping the current attempt");
139 connecting = false;
140 attemptFailed = true; // a late onDisconnect for it isn't a new failure
141 client.disconnect(true);
142 }
143 reconnect.reset(now);
144 }
145
146 if (connecting && static_cast<int32_t>(now - attemptStartedAt) >= static_cast<int32_t>(CONNECT_TIMEOUT_MS)) {
147 // No onConnect or onDisconnect came: reset the client so connect() works again
148 if (log) log->error("Async connect timed out, resetting the client");
149 connecting = false;
150 client.disconnect(true);
151 if (!attemptFailed) {
152 attemptFailed = true;
153 reconnect.failed(now);
154 }
155 }
156
157 if (!isConnected && !connecting && reconnect.due(now)) {
158 if (log) log->info("Async reconnecting");
159 startAttempt();
160 }
161 }
